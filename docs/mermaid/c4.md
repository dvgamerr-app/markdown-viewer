# Mermaid: C4 Diagrams

```mermaid
C4Context
  title System Context diagram for Internet Banking System
  Enterprise_Boundary(b0, "BankBoundary0") {
    Person(customerA, "Banking Customer A", "A customer of the bank.")
    Person_Ext(customerB, "Banking Customer B")
    System(SystemAA, "Internet Banking System", "Allows customers to view accounts.")
    Enterprise_Boundary(b1, "BankBoundary") {
      SystemDb_Ext(SystemE, "Mainframe Banking System", "Stores all core banking info.")
      System_Boundary(b2, "BankBoundary2") {
        System(SystemA, "Banking System A")
        System(SystemB, "Banking System B", "A system of the bank.")
      }
      System_Ext(SystemC, "E-mail system", "Microsoft Exchange")
      SystemQueue(SystemD, "Banking System D Queue", "A system of the bank.")
    }
  }
  BiRel(customerA, SystemAA, "Uses")
  Rel(SystemAA, SystemE, "Uses")
  Rel_U(SystemAA, SystemC, "Sends e-mails", "SMTP")
  Rel_D(SystemC, customerA, "Sends e-mails to")
  UpdateRelStyle(customerA, SystemAA, $textColor="blue", $lineColor="blue", $offsetX="5")
  UpdateLayoutConfig($c4ShapeInRow="3", $c4BoundaryInRow="1")
```
